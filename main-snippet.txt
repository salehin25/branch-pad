63	  const s = String(value ?? '');
64	  if (/[&|<>^`"']/.test(s)) throw new Error(`Unsafe argument: ${s}`);
65	  return s;
66	}

68	function runGh(args) {
69	  const cmd = ['gh', ...args].join(' ');
70	  return execSync(cmd, { encoding: 'utf-8', timeout: 30000 }).trim();
71	}

73	function enrichPr(pr) {
74	  if (!pr) return null;
75	  const reviewers = {};
76	  const reviews = Array.isArray(pr.reviews) ? pr.reviews : [];
77	  for (const r of reviews) {
78	    if (r.user && r.user.login) reviewers[r.user.login] = r.state || '';
79	  }
80	  const approvalsCount = Object.values(reviewers).filter((s) => s === 'APPROVED').length;
81	  const comments = Array.isArray(pr.comments) ? pr.comments : [];
82	  const pendingCommentsCount =
83	    comments.length + reviews.filter((r) => r.state === 'CHANGES_REQUESTED').length;
84	
85	  return {
86	    github_id: pr.id ?? null,
87	    number: pr.number,
88	    title: pr.title || '',
89	    state: pr.state || 'open',
90	    is_draft: !!pr.isDraft,
91	    review_decision: pr.reviewDecision || null,
92	    approvals_count: approvalsCount,
93	    pending_comments_count: pendingCommentsCount,
94	    html_url: pr.url || '',
95	    merged_at: pr.mergedAt || null,
96	  };
97	}
